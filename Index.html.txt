<script>

const input = document.getElementById("input");
const messages = document.getElementById("messages");
const welcome = document.getElementById("welcome");
const history = document.getElementById("history");

function handleKey(event) {
    if (event.key === "Enter" && !event.shiftKey) {
        event.preventDefault();
        sendMessage();
    }
}

function sendMessage() {

    const text = input.value.trim();

    if (!text) return;

    welcome.style.display = "none";

    addMessage(text, "user");

    input.value = "";

    const loading = document.createElement("div");
    loading.className = "message";
    loading.id = "loading";

    loading.innerHTML = `
        <div class="avatar">🌙</div>
        <div class="loading">
            <div class="dot"></div>
            <div class="dot"></div>
            <div class="dot"></div>
        </div>
    `;

    messages.appendChild(loading);
    scrollBottom();

    setTimeout(() => {

        loading.remove();

        const response = getResponse(text);

        addMessage(response, "ai");

    }, 1000);
}

function addMessage(text, type) {

    const message = document.createElement("div");

    message.className = "message " + type;

    message.innerHTML = `
        <div class="avatar">
            ${type === "user" ? "🙂" : "🌙"}
        </div>
        <div class="message-text"></div>
    `;

    message.querySelector(".message-text").textContent = text;

    messages.appendChild(message);

    scrollBottom();
}

function getResponse(text) {

    const lower = text.toLowerCase();

    if (lower.includes("hello") ||
        lower.includes("hi") ||
        lower.includes("hey")) {

        return "Hey! 👋 I'm Lunet. What are we working on?";
    }

    if (lower.includes("who are you") ||
        lower.includes("what are you")) {

        return "I'm Lunet 🌙✨ — your AI assistant.";
    }

    if (lower.includes("how are you")) {

        return "I'm doing great! 🤖✨ Thanks for asking.";
    }

    if (lower.includes("bye")) {

        return "See you later! 🌙";
    }

    return "That's interesting 👀. I'm still learning, but I'll get smarter as we build Lunet!";
}

function newChat() {

    messages.innerHTML = "";
    welcome.style.display = "block";
    input.focus();

}

function scrollBottom() {

    const chat = document.getElementById("chat");

    setTimeout(() => {
        chat.scrollTop = chat.scrollHeight;
    }, 50);

}

input.addEventListener("input", function() {

    this.style.height = "auto";
    this.style.height = Math.min(this.scrollHeight, 160) + "px";

});

</script>
